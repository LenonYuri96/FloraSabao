<?php
/**
 * Flora Sabão — API de Gestão de Estoque & Matérias-Primas
 */
declare(strict_types=1);

require_once __DIR__ . '/../config/database.php';
header('Content-Type: application/json; charset=utf-8');

$action = $_GET['action'] ?? $_POST['action'] ?? '';

try {
    switch ($action) {
        // Retorna todos os insumos com status de estoque (Normal, Alerta ou Crítico)
        case 'listar_insumos':
            $stmt = $pdo->query("
                SELECT id, codigo, nome, unidade, estoque_atual, estoque_minimo, custo_unitario,
                       (estoque_atual * custo_unitario) AS valor_imobilizado,
                       CASE 
                           WHEN estoque_atual <= (estoque_minimo * 0.5) THEN 'CRITICO'
                           WHEN estoque_atual <= estoque_minimo THEN 'ALERTA'
                           ELSE 'NORMAL'
                       END AS status_nivel
                FROM insumos
                ORDER BY nome ASC
            ");
            echo json_encode(['success' => true, 'data' => $stmt->fetchAll()]);
            break;

        // Retorna apenas insumos que necessitam de pedido de compra urgente
        case 'insumos_criticos':
            $stmt = $pdo->query("
                SELECT codigo, nome, unidade, estoque_atual, estoque_minimo,
                       (estoque_minimo - estoque_atual) AS necessidade_compra
                FROM insumos
                WHERE estoque_atual <= estoque_minimo
                ORDER BY (estoque_atual / NULLIF(estoque_minimo, 0)) ASC
            ");
            echo json_encode(['success' => true, 'data' => $stmt->fetchAll()]);
            break;

        // Consulta saldo em tempo real de um insumo específico
        case 'consultar_saldo':
            $id = (int) ($_GET['id'] ?? 0);
            if ($id <= 0) {
                echo json_encode(['success' => false, 'message' => 'ID do insumo inválido.']);
                exit;
            }

            $stmt = $pdo->prepare("SELECT id, codigo, nome, unidade, estoque_atual, estoque_minimo FROM insumos WHERE id = ?");
            $stmt->execute([$id]);
            $insumo = $stmt->fetch();

            if (!$insumo) {
                echo json_encode(['success' => false, 'message' => 'Insumo não encontrado.']);
                exit;
            }

            echo json_encode(['success' => true, 'insumo' => $insumo]);
            break;

        default:
            echo json_encode(['success' => false, 'message' => 'Ação não suportada pelo endpoint de estoque.']);
            break;
    }
} catch (Throwable $e) {
    http_response_code(500);
    echo json_encode(['success' => false, 'message' => 'Erro interno: ' . $e->getMessage()]);
}

<?php
/**
 * Flora Sabão — API de Engenharia de Produtos & Ficha Técnica (BOM)
 */
declare(strict_types=1);

require_once __DIR__ . '/../config/database.php';
header('Content-Type: application/json; charset=utf-8');

$action = $_GET['action'] ?? $_POST['action'] ?? '';

try {
    switch ($action) {
        // Retorna todos os produtos ativos com custos de MP calculados dinamicamente
        case 'listar_produtos':
            $stmt = $pdo->query("
                SELECT p.id, p.codigo, p.nome, p.aroma, p.intensidade_aroma,
                       p.peso_barra_g, p.estoque_atual, p.estoque_minimo, p.preco_venda,
                       COALESCE(SUM(ft.qtd_por_barra * i.custo_unitario), 0) AS custo_mp_unitario
                FROM produtos p
                LEFT JOIN ficha_tecnica ft ON p.id = ft.produto_id
                LEFT JOIN insumos i ON ft.insumo_id = i.id
                WHERE p.ativo = 1
                GROUP BY p.id
                ORDER BY p.codigo ASC
            ");
            echo json_encode(['success' => true, 'data' => $stmt->fetchAll()]);
            break;

        // Detalhes completos da Ficha Técnica de um produto específico
        case 'ficha_tecnica':
            $produto_id = (int) ($_GET['produto_id'] ?? 0);
            if ($produto_id <= 0) {
                echo json_encode(['success' => false, 'message' => 'Produto não especificado.']);
                exit;
            }

            $stmt_prod = $pdo->prepare("SELECT id, codigo, nome, aroma, preco_venda FROM produtos WHERE id = ?");
            $stmt_prod->execute([$produto_id]);
            $produto = $stmt_prod->fetch();

            if (!$produto) {
                echo json_encode(['success' => false, 'message' => 'Produto não encontrado.']);
                exit;
            }

            $stmt_bom = $pdo->prepare("
                SELECT ft.id AS bom_id, ft.qtd_por_barra,
                       i.id AS insumo_id, i.codigo AS insumo_codigo, i.nome AS insumo_nome,
                       i.unidade, i.custo_unitario, i.estoque_atual,
                       (ft.qtd_por_barra * i.custo_unitario) AS subtotal_custo
                FROM ficha_tecnica ft
                JOIN insumos i ON ft.insumo_id = i.id
                WHERE ft.produto_id = ?
                ORDER BY i.nome ASC
            ");
            $stmt_bom->execute([$produto_id]);
            $materiais = $stmt_bom->fetchAll();

            $custo_total_mp = array_sum(array_column($materiais, 'subtotal_custo'));

            echo json_encode([
                'success' => true,
                'produto' => $produto,
                'custo_mp_barra' => $custo_total_mp,
                'margem_bruta' => (float) $produto['preco_venda'] - $custo_total_mp,
                'insumos' => $materiais
            ]);
            break;

        default:
            echo json_encode(['success' => false, 'message' => 'Ação não reconhecida.']);
            break;
    }
} catch (Throwable $e) {
    http_response_code(500);
    echo json_encode(['success' => false, 'message' => 'Erro interno: ' . $e->getMessage()]);
}

document.addEventListener("DOMContentLoaded", () => {
  // 1. Fechar alertas flash de sucesso automaticamente após 4 segundos
  const alertas = document.querySelectorAll(".alert-success");
  alertas.forEach((alerta) => {
    setTimeout(() => {
      const bsAlert = bootstrap.Alert.getOrCreateInstance(alerta);
      if (bsAlert) {
        bsAlert.close();
      }
    }, 4000);
  });

  // 2. Confirmação genérica para links com atributo data-confirm
  document.querySelectorAll("[data-confirm]").forEach((el) => {
    el.addEventListener("click", (e) => {
      const mensagem =
        el.getAttribute("data-confirm") || "Deseja realmente prosseguir?";
      if (!confirm(mensagem)) {
        e.preventDefault();
      }
    });
  });

  // 3. Inicializar Tooltips do Bootstrap
  const tooltipTriggerList = [].slice.call(
    document.querySelectorAll('[data-bs-toggle="tooltip"]'),
  );
  tooltipTriggerList.map(
    (tooltipTriggerEl) => new bootstrap.Tooltip(tooltipTriggerEl),
  );
});

/**
 * Flora Sabão — Lógica de Chão de Fábrica e Ordens de Produção (op.js)
 */

document.addEventListener("DOMContentLoaded", () => {
  // 1. Controle do Modal de Apontamento Final de Produção
  const modalApontarEl = document.getElementById("modalApontar");
  if (modalApontarEl) {
    const modalApontar = bootstrap.Modal.getOrCreateInstance(modalApontarEl);
    const apontarOpId = document.getElementById("apontar_op_id");
    const apontarOpNum = document.getElementById("apontar_op_num");
    const apontarOpQtd = document.getElementById("apontar_op_qtd");
    const apontarBoas = document.getElementById("apontar_boas");
    const apontarRefugo = document.getElementById("apontar_refugo");

    document.querySelectorAll(".btn-apontar").forEach((btn) => {
      btn.addEventListener("click", () => {
        const opId = btn.getAttribute("data-op-id");
        const opNum = btn.getAttribute("data-op-num");
        const qtd = btn.getAttribute("data-qtd");

        apontarOpId.value = opId;
        apontarOpNum.textContent = opNum;
        apontarOpQtd.textContent = qtd;
        apontarBoas.value = qtd;
        apontarRefugo.value = 0;

        modalApontar.show();
      });
    });
  }

  // 2. Cálculo e Renderização Dinâmica da Explosão de BOM
  const modalBomEl = document.getElementById("modalBomExplosion");
  if (modalBomEl) {
    const modalBom = bootstrap.Modal.getOrCreateInstance(modalBomEl);
    const container = document.getElementById("conteudoBomExplosion");

    document.querySelectorAll(".btn-ver-bom").forEach((btn) => {
      btn.addEventListener("click", () => {
        const prodId = btn.getAttribute("data-prod-id");
        const opQtd = parseInt(btn.getAttribute("data-qtd"), 10) || 1;
        const opNum = btn.getAttribute("data-op-num");

        container.innerHTML = `
          <div class="alert alert-secondary py-2 small mb-3">
            Simulação MRP para <strong>${opNum}</strong>: Lote de <strong>${opQtd} barras</strong>.
          </div>
          <div class="text-center py-4">
            <div class="spinner-border text-success" role="status"></div>
            <p class="small text-muted mt-2">Calculando explosão de materiais...</p>
          </div>
        `;

        modalBom.show();

        // Requisição para a API
        const apiUrl = `../api/op_api.php?action=get_bom&produto_id=${prodId}&qtd=${opQtd}`;
        fetch(apiUrl)
          .then((res) => {
            if (!res.ok) throw new Error("Erro na resposta da rede");
            return res.json();
          })
          .then((data) => {
            if (!data.success || !data.items || data.items.length === 0) {
              container.innerHTML = `
                <div class="alert alert-warning py-3">
                  <i class="bi bi-exclamation-triangle-fill me-2"></i>
                  Não há Ficha Técnica (BOM) cadastrada para este produto.
                </div>
              `;
              return;
            }

            let linhas = "";
            let temRuptura = false;

            data.items.forEach((item) => {
              const nec = parseFloat(item.qtd_por_barra) * opQtd;
              const estoque = parseFloat(item.estoque_atual);
              const saldo = estoque - nec;
              const falta = saldo < 0;

              if (falta) temRuptura = true;

              linhas += `
                <tr class="${falta ? "table-danger" : ""}">
                  <td><strong>${item.nome}</strong></td>
                  <td class="font-monospace">${parseFloat(item.qtd_por_barra).toFixed(4)} ${item.unidade}</td>
                  <td class="font-monospace fw-bold">${nec.toFixed(4)} ${item.unidade}</td>
                  <td class="font-monospace">${estoque.toFixed(4)} ${item.unidade}</td>
                  <td class="font-monospace fw-bold ${falta ? "text-danger" : "text-success"}">
                    ${saldo.toFixed(4)} ${item.unidade}
                  </td>
                  <td>
                    ${
                      falta
                        ? '<span class="badge bg-danger"><i class="bi bi-x-circle me-1"></i> Ruptura</span>'
                        : '<span class="badge bg-success"><i class="bi bi-check-circle me-1"></i> Suficiente</span>'
                    }
                  </td>
                </tr>
              `;
            });

            container.innerHTML = `
              <div class="alert ${temRuptura ? "alert-danger" : "alert-success"} py-2 small mb-3">
                ${
                  temRuptura
                    ? '<i class="bi bi-exclamation-octagon-fill me-1"></i> <strong>Risco de Parada:</strong> Um ou mais insumos não atendem à quantidade planejada.'
                    : '<i class="bi bi-check-circle-fill me-1"></i> <strong>Tudo certo:</strong> Há matérias-primas suficientes para rodar este lote.'
                }
              </div>
              <div class="table-responsive">
                <table class="table table-sm table-bordered align-middle mb-0">
                  <thead class="table-light small text-uppercase">
                    <tr>
                      <th>Insumo</th>
                      <th>Consumo/Barra</th>
                      <th>Demanda Total</th>
                      <th>Estoque Atual</th>
                      <th>Saldo Projetado</th>
                      <th>Status PCP</th>
                    </tr>
                  </thead>
                  <tbody>${linhas}</tbody>
                </table>
              </div>
            `;
          })
          .catch(() => {
            container.innerHTML = `
              <div class="alert alert-danger py-2 small">
                Erro ao calcular a explosão de materiais. Verifique a conexão com o servidor.
              </div>
            `;
          });
      });
    });
  }
});

# ------------------------------------------------------------------
# Funções auxiliares

def eh_arquivo_texto(caminho: Path) -> bool:
    """Verifica se o arquivo tem extensão de texto/código."""
    return caminho.suffix.lower() in EXTENSOES_TEXTO or caminho.name in {
        ".htaccess",
        ".gitignore",
        ".env",
    }


def ler_conteudo(caminho: Path) -> str:
    """
    Lê o conteúdo do arquivo tentando múltiplos encodings.
    Retorna string vazia se não conseguir ler.
    """
    encodings = ["utf-8", "utf-8-sig", "latin-1", "cp1252"]
    for enc in encodings:
        try:
            return caminho.read_text(encoding=enc)
        except (UnicodeDecodeError, UnicodeError):
            continue
        except Exception as e:
            return f"[ERRO AO LER: {e}]"
    return "[ERRO: encoding não reconhecido]"


def deve_ignorar(caminho: Path) -> bool:
    """Verifica se o caminho (pasta ou arquivo) deve ser ignorado."""
    partes = set(caminho.parts)
    if partes & PASTAS_IGNORADAS:
        return True
    if caminho.name in ARQUIVOS_IGNORADOS:
        return True
    return False


# ------------------------------------------------------------------
# Coleta de arquivos
# ------------------------------------------------------------------
def coletar_arquivos(base: Path):
    """
    Percorre recursivamente a pasta e retorna uma lista de dicionários:
        [{"path": "api/op_api.php", "ext": ".php", "conteudo": "...", "linhas": N, "bytes": M}, ...]
    """
    arquivos = []
    for caminho in sorted(base.rglob("*")):
        if not caminho.is_file():
            continue
        if deve_ignorar(caminho):
            continue
        if not eh_arquivo_texto(caminho):
            continue

        relativo = caminho.relative_to(base).as_posix()
        conteudo = ler_conteudo(caminho)
        arquivos.append(
            {
                "path": relativo,
                "ext": caminho.suffix.lower(),
                "conteudo": conteudo,
                "linhas": conteudo.count("\n") + 1 if conteudo else 0,
                "bytes": caminho.stat().st_size,
            }
        )
    return arquivos

def gerar_txt(arquivos, base):
    linhas = []
    linhas.append("=" * 78)
    linhas.append(f" EXTRAÇÃO DE CÓDIGOS/ALGORITMOS — PROJETO: {base.name}")
    linhas.append(f" Gerado em: {datetime.now().strftime('%d/%m/%Y %H:%M:%S')}")
    linhas.append(f" Total de arquivos: {len(arquivos)}")
    linhas.append("=" * 78)
    linhas.append("")

    # Sumário
    linhas.append("📂 SUMÁRIO DE ARQUIVOS")
    linhas.append("-" * 78)
    for a in arquivos:
        linhas.append(
            f"  {a['path']:<45}  {a['linhas']:>4} linhas  {a['bytes']:>7} bytes"
        )
    linhas.append("")

    # Conteúdo
    for i, a in enumerate(arquivos, 1):
        linhas.append("=" * 78)
        linhas.append(f"📄 [{i}/{len(arquivos)}] {a['path']}")
        linhas.append(
            f"    Extensão: {a['ext'] or '(sem extensão)'} | Linhas: {a['linhas']} | Bytes: {a['bytes']}"
        )
        linhas.append("=" * 78)
        linhas.append(a["conteudo"].rstrip())
        linhas.append("")
        linhas.append("")

    return "\n".join(linhas)

    # Conteúdo
    for i, a in enumerate(arquivos, 1):

def gerar_json(arquivos, base):
    dados = {
        "projeto": base.name,
        "gerado_em": datetime.now().isoformat(),
        "total_arquivos": len(arquivos),
        "arquivos": [
            {
                "path": a["path"],
                "extensao": a["ext"],
                "linhas": a["linhas"],
                "bytes": a["bytes"],
                "conteudo": a["conteudo"],
            }
            for a in arquivos
        ],
    }
    return json.dumps(dados, ensure_ascii=False, indent=2)

def main():
    parser = argparse.ArgumentParser(
        description="Extrai códigos/algoritmos e caminhos do projeto Flora Sabão."
    )
    parser.add_argument(
        "--pasta",
        default=PASTA_PROJETO_PADRAO,
        help=f"Pasta do projeto (padrão: {PASTA_PROJETO_PADRAO})",
    )
    parser.add_argument(
        "--formato",
        default=FORMATO_PADRAO,
        choices=["txt", "md", "json"],
        help=f"Formato de saída (padrão: {FORMATO_PADRAO})",
    )
    parser.add_argument(
        "--saida",
        default=None,
        help="Nome do arquivo de saída (sem extensão, opcional)",
    )
    args = parser.parse_args()

    base = Path(args.pasta)
    if not base.exists() or not base.is_dir():
        print(f"❌ Pasta não encontrada: {base.resolve()}")
        return

    print(f"\n🔍 Varrendo: {base.resolve()}")
    arquivos = coletar_arquivos(base)

    if not arquivos:
        print("⚠  Nenhum arquivo de texto encontrado.")
        return

    print(f"✔ {len(arquivos)} arquivo(s) encontrado(s).\n")

    # Gera o conteúdo conforme o formato
    if args.formato == "txt":
        conteudo_saida = gerar_txt(arquivos, base)
    elif args.formato == "md":
        conteudo_saida = gerar_md(arquivos, base)
    else:
        conteudo_saida = gerar_json(arquivos, base)

    # Nome do arquivo de saída
    nome_saida = args.saida or f"{ARQUIVO_SAIDA_PADRAO}.{args.formato}"
    caminho_saida = Path(nome_saida)

    caminho_saida.write_text(conteudo_saida, encoding="utf-8")

    tamanho_kb = caminho_saida.stat().st_size / 1024
    print(f"✅ Relatório gerado: {caminho_saida.resolve()}")
    print(f"   • Formato:  {args.formato}")
    print(f"   • Tamanho:  {tamanho_kb:.1f} KB")
    print(f"   • Arquivos: {len(arquivos)}\n")


if __name__ == "__main__":
    main()

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $acao = $_POST['acao'] ?? '';

    // 1. CRIAR NOVA OP (Planejamento)
    if ($acao === 'criar_op') {
        $produto_id = (int) ($_POST['produto_id'] ?? 0);
        $qtd_planejada = (int) ($_POST['quantidade_planejada'] ?? 0);
        $data_prog = $_POST['data_programada'] ?? '';
        $obs = trim($_POST['observacoes'] ?? '');

        if ($produto_id <= 0 || $qtd_planejada <= 0 || empty($data_prog)) {
            $mensagem_erro = 'Selecione o produto, informe quantidade positiva e data válida.';
        } else {
            // Gerar número sequencial de OP (ex: OP-2026-004)
            $ano = date('Y', strtotime($data_prog));
            $stmt_count = $pdo->query("SELECT COUNT(*) + 1 FROM ordens_producao");
            $seq = str_pad((string) $stmt_count->fetchColumn(), 3, '0', STR_PAD_LEFT);
            $numero_op = "OP-{$ano}-{$seq}";

            $stmt = $pdo->prepare("
                INSERT INTO ordens_producao (numero, produto_id, quantidade_planejada, data_programada, status, observacoes)
                VALUES (?, ?, ?, ?, 'planejada', ?)
            ");
            $stmt->execute([$numero_op, $produto_id, $qtd_planejada, $data_prog, $obs]);
            $mensagem_sucesso = "Ordem de Produção {$numero_op} criada com sucesso!";
        }
    }

    // 2. INICIAR PRODUÇÃO (Liberar chão de fábrica)
    if ($acao === 'iniciar_op') {
        $op_id = (int) ($_POST['op_id'] ?? 0);
        $stmt = $pdo->prepare("UPDATE ordens_producao SET status = 'em_producao', data_inicio = NOW() WHERE id = ? AND status = 'planejada'");
        $stmt->execute([$op_id]);
        $mensagem_sucesso = 'Status da OP atualizado para "Em Produção".';
    }

    // 3. FINALIZAR OP (Apontamento de Boas + Refugo e Baixa Automática de Insumos)
    if ($acao === 'finalizar_op') {
        $op_id = (int) ($_POST['op_id'] ?? 0);
        $qtd_produzida = (int) ($_POST['quantidade_produzida'] ?? 0);
        $qtd_refugo = (int) ($_POST['quantidade_refugo'] ?? 0);

        if ($op_id <= 0 || ($qtd_produzida + $qtd_refugo) <= 0) {
            $mensagem_erro = 'A soma de barras boas e refugo deve ser maior que zero.';
        } else {
            try {
                $pdo->beginTransaction();

                // 3.1 Busca a OP com lock
                $stmt_op = $pdo->prepare("SELECT * FROM ordens_producao WHERE id = ? FOR UPDATE");
                $stmt_op->execute([$op_id]);
                $op = $stmt_op->fetch();

                if (!$op || $op['status'] === 'finalizada') {
                    throw new Exception('OP não encontrada ou já finalizada.');
                }

                $total_processado = $qtd_produzida + $qtd_refugo;

                // 3.2 Explosão da BOM para consumir os insumos do total processado (Boas + Perdas)
                $stmt_bom = $pdo->prepare("
                    SELECT ft.insumo_id, ft.qtd_por_barra, i.nome, i.estoque_atual, i.unidade
                    FROM ficha_tecnica ft
                    JOIN insumos i ON ft.insumo_id = i.id
                    WHERE ft.produto_id = ?
                ");
                $stmt_bom->execute([$op['produto_id']]);
                $ingredientes = $stmt_bom->fetchAll();

                if (empty($ingredientes)) {
                    throw new Exception('Esta fragrância não possui Ficha Técnica cadastrada. Configure a BOM antes de finalizar.');
                }

                // 3.3 Consome os insumos e gera registro no Kardex
                foreach ($ingredientes as $ing) {
                    $qtd_consumida = $ing['qtd_por_barra'] * $total_processado;

                    // Baixa o estoque do insumo
                    $stmt_baixa = $pdo->prepare("UPDATE insumos SET estoque_atual = estoque_atual - ? WHERE id = ?");
                    $stmt_baixa->execute([$qtd_consumida, $ing['insumo_id']]);

                    // Registra saída no histórico
                    $stmt_mov_ins = $pdo->prepare("
                        INSERT INTO mov_estoque (tipo, tipo_item, insumo_id, quantidade, op_id, motivo)
                        VALUES ('saida', 'insumo', ?, ?, ?, ?)
                    ");
                    $stmt_mov_ins->execute([$ing['insumo_id'], $qtd_consumida, $op_id, "Consumo Produção {$op['numero']}"]);
                }

                // 3.4 Incrementa o estoque do produto acabado (apenas barras boas aprovadas)
                if ($qtd_produzida > 0) {
                    $stmt_prod_up = $pdo->prepare("UPDATE produtos SET estoque_atual = estoque_atual + ? WHERE id = ?");
                    $stmt_prod_up->execute([$qtd_produzida, $op['produto_id']]);

                    // Registra entrada de produto acabado no histórico
                    $stmt_mov_p = $pdo->prepare("
                        INSERT INTO mov_estoque (tipo, tipo_item, produto_id, quantidade, op_id, motivo)
                        VALUES ('entrada', 'produto', ?, ?, ?, ?)
                    ");
                    $stmt_mov_p->execute([$op['produto_id'], $qtd_produzida, $op_id, "Entrada PA Lote {$op['numero']}"]);
                }

                // 3.5 Atualiza a OP
                $stmt_finaliza = $pdo->prepare("
                    UPDATE ordens_producao 
                    SET quantidade_produzida = ?, quantidade_refugo = ?, data_finalizacao = NOW(), status = 'finalizada'
                    WHERE id = ?
                ");
                $stmt_finaliza->execute([$qtd_produzida, $qtd_refugo, $op_id]);

                $pdo->commit();
                $mensagem_sucesso = "OP {$op['numero']} finalizada! Foram estocadas {$qtd_produzida} barras e os insumos foram baixados automaticamente.";
            } catch (Exception $e) {
                if ($pdo->inTransaction()) {
                    $pdo->rollBack();
                }
                $mensagem_erro = $e->getMessage();
            }
        }
    }

    // 4. CANCELAR OP
    if ($acao === 'cancelar_op') {
        $op_id = (int) ($_POST['op_id'] ?? 0);
        $stmt = $pdo->prepare("UPDATE ordens_producao SET status = 'cancelada' WHERE id = ? AND status != 'finalizada'");
        $stmt->execute([$op_id]);
        $mensagem_sucesso = 'Ordem de Produção cancelada.';
    }

$escala_aroma = [
    'suave' => 1, // Ex: Lavanda
    'moderado' => 2, // Ex: Pitanga, Rosas
    'intenso' => 3  // Ex: Maracujá cítrico
];

// ---------------------------------------------------------
// REORDENAÇÃO AUTOMÁTICA OTIMIZADA (Heurística de Menor Setup)
// ---------------------------------------------------------
if ($_SERVER['REQUEST_METHOD'] === 'POST' && ($_POST['acao'] ?? '') === 'otimizar_sequencia') {
    // Busca todas as OPs ativas
    $stmt = $pdo->query("
        SELECT op.id, p.intensidade_aroma, p.codigo
        FROM ordens_producao op
        JOIN produtos p ON op.produto_id = p.id
        WHERE op.status IN ('planejada', 'em_producao')
    ");
    $ops_ativas = $stmt->fetchAll();

    // Ordena da menor intensidade para a maior (Suave -> Moderado -> Intenso)
    usort($ops_ativas, function ($a, $b) use ($escala_aroma) {
        $nivelA = $escala_aroma[$a['intensidade_aroma']] ?? 2;
        $nivelB = $escala_aroma[$b['intensidade_aroma']] ?? 2;
        if ($nivelA === $nivelB) {
            return strcmp($a['codigo'], $b['codigo']);
        }
        return $nivelA <=> $nivelB;
    });

    // Atualiza as datas programadas sequencialmente em passos diários
    $data_base = new DateTime();
    foreach ($ops_ativas as $item) {
        $stmt_up = $pdo->prepare("UPDATE ordens_producao SET data_programada = ? WHERE id = ?");
        $stmt_up->execute([$data_base->format('Y-m-d'), $item['id']]);
        $data_base->modify('+1 day');
    }

    $mensagem_sucesso = 'Heurística aplicada com sucesso! Lotes reordenados do aroma suave ao mais intenso para minimizar o tempo de setup.';
}

# 🌿 Flora Sabão — Simulador Didático de PCP

## 🎯 Objetivos de Aprendizagem

- **Explosão de Materiais (MRP I / BOM):** Cálculo de necessidades líquidas lote a lote a partir de fórmulas unitárias.
- **Risco de Contaminação Cruzada & Matriz de Setup:** Identificação de gargalos na troca de fragrâncias e sequenciamento ótimo (_Aroma Suave &rarr; Aroma Intenso_).
- **Apontamento de Chão de Fábrica:** Registro de barras boas vs. refugo com baixa automática proporcional de matérias-primas no estoque.
- **Indicadores de Desempenho (KPIs):** Apuração prática de **OTIF**, **Aderência ao Plano**, **Taxa de Refugo** e **Ocupação da Capacidade**.

-- --------------------------------------------------------
-- Tabela: PRODUTOS (Fragrâncias acabadas)
-- --------------------------------------------------------
CREATE TABLE `produtos` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `codigo` VARCHAR(10) NOT NULL UNIQUE,
    `nome` VARCHAR(50) NOT NULL,
    `aroma` VARCHAR(50) NOT NULL,
    `peso_barra_g` DECIMAL(6, 2) NOT NULL DEFAULT 100.00,
    `intensidade_aroma` ENUM('suave', 'moderado', 'intenso') NOT NULL DEFAULT 'moderado',
    `estoque_atual` INT NOT NULL DEFAULT 0,
    `estoque_minimo` INT NOT NULL DEFAULT 50,
    `preco_venda` DECIMAL(10, 2) NOT NULL DEFAULT 14.90,
    `ativo` TINYINT(1) NOT NULL DEFAULT 1
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4;

-- --------------------------------------------------------
-- Tabela: INSUMOS (Matéria-prima e embalagens)
-- --------------------------------------------------------
CREATE TABLE `insumos` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `codigo` VARCHAR(10) NOT NULL UNIQUE,
    `nome` VARCHAR(80) NOT NULL,
    `unidade` VARCHAR(10) NOT NULL,
    -- 'kg', 'L', 'un'
    `estoque_atual` DECIMAL(10, 4) NOT NULL DEFAULT 0.0000,
    `estoque_minimo` DECIMAL(10, 4) NOT NULL DEFAULT 0.0000,
    `custo_unitario` DECIMAL(10, 2) NOT NULL DEFAULT 0.00
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4;

-- --------------------------------------------------------
-- Tabela: FICHA TÉCNICA (Bill of Materials - BOM por barra)
-- --------------------------------------------------------
CREATE TABLE `ficha_tecnica` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `produto_id` INT NOT NULL,
    `insumo_id` INT NOT NULL,
    `qtd_por_barra` DECIMAL(10, 4) NOT NULL,
    -- Consumo unitário
    FOREIGN KEY (`produto_id`) REFERENCES `produtos`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`insumo_id`) REFERENCES `insumos`(`id`) ON DELETE RESTRICT,
    UNIQUE KEY `uk_prod_insumo` (`produto_id`, `insumo_id`)
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4;

-- --------------------------------------------------------
-- Tabela: ORDENS DE PRODUÇÃO (OP)
-- --------------------------------------------------------
CREATE TABLE `ordens_producao` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `numero` VARCHAR(20) NOT NULL UNIQUE,
    `produto_id` INT NOT NULL,
    `quantidade_planejada` INT NOT NULL,
    `quantidade_produzida` INT NOT NULL DEFAULT 0,
    `quantidade_refugo` INT NOT NULL DEFAULT 0,
    `data_programada` DATE NOT NULL,
    `data_inicio` DATETIME NULL,
    `data_finalizacao` DATETIME NULL,
    `status` ENUM(
        'planejada',
        'em_producao',
        'finalizada',
        'cancelada'
    ) NOT NULL DEFAULT 'planejada',
    `observacoes` TEXT NULL,
    `created_at` DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (`produto_id`) REFERENCES `produtos`(`id`)
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4;

-- --------------------------------------------------------
-- Tabela: MOVIMENTAÇÃO DE ESTOQUE
-- --------------------------------------------------------
CREATE TABLE `mov_estoque` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `tipo` ENUM('entrada', 'saida') NOT NULL,
    `tipo_item` ENUM('insumo', 'produto') NOT NULL,
    `insumo_id` INT NULL,
    `produto_id` INT NULL,
    `quantidade` DECIMAL(10, 4) NOT NULL,
    `op_id` INT NULL,
    `data_mov` DATETIME DEFAULT CURRENT_TIMESTAMP,
    `motivo` VARCHAR(100) NOT NULL,
    -- 'Produção OP-001', 'Compra NF 123', 'Ajuste inventário'
    FOREIGN KEY (`insumo_id`) REFERENCES `insumos`(`id`),
    FOREIGN KEY (`produto_id`) REFERENCES `produtos`(`id`),
    FOREIGN KEY (`op_id`) REFERENCES `ordens_producao`(`id`)
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4;
